async function fetchCentralData() { const loadingBar = document.getElementById('loadingStatus'); loadingBar.classList.remove('hidden'); try { // 1. Obtener fecha del último commit en GitHub let serverDate = "Local"; try { const commitRes = await fetch(`https://api.github.com/repos/${GITHUB_USER}/${GITHUB_REPO}/commits?page=1&per_page=1`); if (commitRes.ok) { const commitData = await commitRes.json(); if (commitData && commitData.length > 0) { serverDate = commitData[0].commit.committer.date; const dateObj = new Date(serverDate); document.getElementById('lblLastUpdate').innerText = dateObj.toLocaleDateString('es-PE') + ' ' + dateObj.toLocaleTimeString('es-PE', { hour: '2-digit', minute: '2-digit', second: '2-digit' }); } } } catch(e) { console.warn(e); } // 2. Comprobar si ya tenemos los datos guardados en la memoria del navegador const cachedVersion = localStorage.getItem('albrecht_cache_version'); const cachedSummary = localStorage.getItem('albrecht_cached_data'); const cachedRaw = localStorage.getItem('albrecht_cached_raw'); if (cachedVersion === serverDate && cachedSummary && cachedRaw) { // Carga instantánea desde memoria interna loadingBar.innerText = "Cargando desde memoria ultra rápida..."; currentData = JSON.parse(cachedSummary); rawRecordsData = JSON.parse(cachedRaw); populateFilters(); updateDashboard(); loadingBar.classList.add('hidden'); return; } // 3. Si hubo cambios en GitHub, descarga en paralelo los meses loadingBar.innerText = "Sincronizando nuevos registros desde GitHub..."; currentData = []; rawRecordsData = []; globalSeenActos.clear(); const monthFiles = [ 'mes_01.txt', 'mes_02.txt', 'mes_03.txt', 'mes_04.txt', 'mes_05.txt', 'mes_06.txt', 'mes_07.txt', 'mes_08.txt', 'mes_09.txt', 'mes_10.txt', 'mes_11.txt', 'mes_12.txt' ]; // Descarga todos a la vez en paralelo const fetchPromises = monthFiles.map(file => fetch(`https://raw.githubusercontent.com/${GITHUB_USER}/${GITHUB_REPO}/main/data/${file}?t=${Date.now()}`) .then(r => r.ok ? r.text() : null) .catch(() => null) ); const textResults = await Promise.all(fetchPromises); let totalCargados = 0; textResults.forEach(text => { if (!text) return; const lines = text.split(/\r\n|\n/); if (lines.length > 1) { let delim = lines[0].includes('|') ? '|' : (lines[0].includes(';') ? ';' : ','); const rows = lines.map(line => line.split(delim).map(c => c.trim())); totalCargados += parseExcelToSummary(rows); } }); // 4. Guardar en memoria para que las próximas aperturas sean inmediatas if (totalCargados > 0) { try { localStorage.setItem('albrecht_cache_version', serverDate); localStorage.setItem('albrecht_cached_data', JSON.stringify(currentData)); localStorage.setItem('albrecht_cached_raw', JSON.stringify(rawRecordsData)); } catch(e) { console.warn("Límite de memoria local alcanzado:", e); } populateFilters(); updateDashboard(); loadingBar.classList.add('hidden'); } else { loadingBar.classList.remove('bg-cyan-600', 'animate-pulse'); loadingBar.classList.add('bg-rose-600'); loadingBar.innerText = "No se encontraron los datos en GitHub."; } } catch(err) { console.error(err); loadingBar.innerText = "Error cargando datos: " + err.message; } }
